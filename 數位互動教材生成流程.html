做得好的事

  • 單一 HTML 檔案，零依賴，易部署
  • CSS 自訂屬性集中管理色彩

踩過的坑（已納入 SOP）

  • Tab 捲軸：用 flex:1 取代 overflow-y:auto
  • Canvas 溢出：flex 子容器加 min-width:0
  • RAF 衝突：切換前先 cancelAnimationFrame
  • GitHub Pages 延遲：告知使用者 Ctrl+Shift+R

每個動畫必須用 IIFE 包裹
// ✅ 正確：變數不會汙染全域
(function() {
  var c = document.getElementById('myCanvas');
  if (!c) return;
  var ctx = c.getContext('2d');
  var W = c.width, H = c.height;
  var t = 0;

  function draw() {
    ctx.clearRect(0, 0, W, H);
    // ... 繪製邏輯
    t++;
    requestAnimationFrame(draw);
  }
  draw();
})();

// ❌ 錯誤：全域變數，多個動畫時 t 會互相覆蓋
var t = 0;
function draw() { ... }
2
必備工具函式（加在 <script> 最上方）
// 畫箭頭
function drawArrow(ctx, x1, y1, x2, y2, color, lw) {
  var angle = Math.atan2(y2-y1, x2-x1), hl = 12;
  ctx.save();
  ctx.strokeStyle = color; ctx.fillStyle = color; ctx.lineWidth = lw||2;
  ctx.beginPath(); ctx.moveTo(x1,y1); ctx.lineTo(x2,y2); ctx.stroke();
  ctx.beginPath();
  ctx.moveTo(x2, y2);
  ctx.lineTo(x2 - hl*Math.cos(angle-Math.PI/6), y2 - hl*Math.sin(angle-Math.PI/6));
  ctx.lineTo(x2 - hl*Math.cos(angle+Math.PI/6), y2 - hl*Math.sin(angle+Math.PI/6));
  ctx.closePath(); ctx.fill();
  ctx.restore();
}

// 圓角矩形 polyfill（舊版 Chrome 沒有 roundRect）
if (!CanvasRenderingContext2D.prototype.roundRect) {
  CanvasRenderingContext2D.prototype.roundRect = function(x,y,w,h,r) {
    this.beginPath();
    this.moveTo(x+r, y);
    this.arcTo(x+w, y, x+w, y+h, r);
    this.arcTo(x+w, y+h, x, y+h, r);
    this.arcTo(x, y+h, x, y, r);
    this.arcTo(x, y, x+w, y, r);
    this.closePath();
  };
}
3
生活應用 Tab 動畫：切換時先取消舊 RAF
var lifeAnimId = null;
var lifeActive = 0;

function showLife(i) {
  lifeActive = i;
  if (lifeAnimId) { cancelAnimationFrame(lifeAnimId); lifeAnimId = null; }
  // 更新 tab 樣式、標題、說明
  document.querySelectorAll('.life-tab-btn').forEach(function(b,j) {
    b.classList.toggle('active', j===i);
  });
  // 啟動對應動畫
  [anim0, anim1, anim2, anim3, anim4, anim5][i]();
}

function anim0() {
  var idx = 0;
  (function loop(t) {
    if (lifeActive !== idx) return; // 守衛：已切換就停止
    lifeAnimId = requestAnimationFrame(function() { loop(t+1); });
    // ... 繪製邏輯
  })(0);
}

版面配置規則
三個常見版面的正確 CSS，避免捲軸與溢出問題
1
實驗：左控制 + 右 Canvas（flex）
.exp-layout { display:flex; gap:16px; align-items:flex-start; }
.exp-controls { flex:0 0 220px; min-width:160px; }
.exp-canvas-wrap {
  flex:1;
  min-width:0; /* 關鍵！沒有這行 Canvas 會撐破容器 */
}
2
生活應用：垂直 Tab 列不出現捲軸
.life-layout {
  display:flex;
  height: clamp(340px, calc(100vh - 220px), 500px);
  overflow:hidden;
}
.life-tab-list {
  display:flex; flex-direction:column; gap:2px; padding:6px 5px;
  min-width:90px; flex-shrink:0;
  overflow:hidden; /* 禁用 overflow-y:auto（會產生捲軸）*/
}
.life-tab-btn {
  flex:1; /* 平均分配高度，6 個按鈕各佔 1/6 */
  min-height:0; /* 允許縮小 */
  display:flex; flex-direction:column;
  align-items:center; justify-content:center;
}
🚫 禁止用 overflow-y:auto 在 Tab 列上，這會產生捲軸。正確做法是讓按鈕用 flex:1 自動縮小。

常見問題速查
遇到問題時第一步就查這裡
現象 原因 解法
改了 HTML，預覽沒變化 瀏覽器快取 location.reload(true)
左側 Tab 列出現捲軸 Tab 按鈕高度固定 加 flex:1; min-height:0 到 .life-tab-btn
Canvas 超出右側邊界 flex 容器沒有 min-width:0 在 Canvas 的 flex 包裹容器加 min-width:0
切換 Tab 時兩個動畫同時跑 沒取消舊的 RAF 切換前 cancelAnimationFrame(lifeAnimId)
頁面有水平捲軸 某元素超出 body 寬度 body { overflow-x:hidden } + 找出超出的元素
GitHub Pages 顯示舊版 CDN 快取延遲 等 1–3 分鐘，再 Ctrl+Shift+R
Canvas 動畫不動 元素 ID 打錯 / 動畫未啟動 確認 document.getElementById('id') 非 null，確認 INIT 段有呼叫
roundRect 報錯 舊版瀏覽器不支援 加 polyfill（見 Phase 4 Step 2）
